Mọi hàm có thể bị yêu cầu cài đặt. Mã theo phong cách giáo trình (LIST pHead/pTail, QuickSort phần tử giữa, Shift/CreateHeap). Toàn bộ đã biên dịch và chạy đúng bằng clang++ (mẫu gốc: /tmp/check.cpp).

Ghi nhớ mỗi hàm kèm một dòng chú thích in nghiêng — đó là bất biến/bẫy mà chấm điểm thường soi (swap vô điều kiện của Selection, pTail khi thêm/xóa cuối, khóa bằng thì không chèn…).

Tìm kiếm tuyến tính — mảng tăng dần

int LinearSearch(int a[], int n, int x) {
    // a[i] > x thì dừng: sau đó không còn x
    for (int i = 0; i < n && a[i] <= x; i++)
        if (a[i] == x) return i;
    return -1;
}

Tìm kiếm tuyến tính — bản thường (0.25đ)

int LinearSearchPlain(int a[], int n, int x) {
    // duyệt hết, không dùng điều kiện tăng dần
    for (int i = 0; i < n; i++)
        if (a[i] == x) return i;
    return -1;
}

Tìm kiếm nhị phân

int BinarySearch(int a[], int n, int x) {
    // left..right thu hẹp một nửa mỗi vòng
    int left = 0, right = n - 1;
    while (left <= right) {
        int mid = (left + right) / 2;
        if (a[mid] == x) return mid;
        if (x < a[mid]) right = mid - 1;
        else left = mid + 1;
    }
    return -1;
}

SelectionSort — O(n²)

void SelectionSort(int *a, const int &n) {
    // swap MỖI vòng i, kể cả min == i (bẫy đếm phép gán!)
    for (int i = 0; i < n - 1; i++) {
        int min = i;
        for (int j = i + 1; j < n; j++)
            if (a[j] < a[min]) min = j;
        swap(a[i], a[min]);
    }
}

InsertionSort — O(n) / O(n²)

void InsertionSort(int a[], int n) {
    // chèn x = a[i] vào đoạn [0..i-1] đãSORT
    for (int i = 1; i < n; i++) {
        int x = a[i], pos = i - 1;
        while (pos >= 0 && a[pos] > x) {
            a[pos + 1] = a[pos]; pos--;
        }
        a[pos + 1] = x;
    }
}

QuickSort — O(n log n) / O(n²)

void QuickSort(int a[], int left, int right) {
    // x = phần tử GIỮA; do-while; swap khi i <= j
    int i = left, j = right, x = a[(left + right) / 2];
    do {
        while (a[i] < x) i++;
        while (a[j] > x) j--;
        if (i <= j) { swap(a[i], a[j]); i++; j--; }
    } while (i < j);
    if (left < j) QuickSort(a, left, j);
    if (i < right) QuickSort(a, i, right);
}

Shift + CreateHeap + HeapSort — O(n log n)

void Shift(int a[], int l, int r) {
    // sàng max-heap (0-indexed); con LỚN hơn mới lên
    int i = l, j = 2 * i + 1, x = a[i];
    while (j <= r) {
        if (j < r && a[j] < a[j + 1]) j++;
        if (a[j] <= x) break;
        a[i] = a[j]; i = j; j = 2 * i + 1;
    }
    a[i] = x;
}
void CreateHeap(int a[], int n) {
    // sàng từ node cuối-có-con: l = n/2 - 1 trở về 0
    for (int l = n / 2 - 1; l >= 0; l--) Shift(a, l, n - 1);
}
void HeapSort(int a[], int n) {
    // swap a[0]↔a[r] rồi sàng lại đoạn [0..r-1]
    CreateHeap(a, n);
    for (int r = n - 1; r > 0; r--) {
        swap(a[0], a[r]); Shift(a, 0, r - 1);
    }
}

DSLK — struct & thao tác cơ bản

struct NODE { int info; NODE *pNext; };
struct LIST { NODE *pHead, *pTail; };

void CreateEmptyList(LIST &L) { L.pHead = L.pTail = NULL; }
NODE *CreateNode(int x) {
    NODE *p = new NODE;
    p->info = x; p->pNext = NULL; return p;
}
void AddHead(LIST &L, NODE *p) {
    // rỗng: pHead = pTail = p
    if (!L.pHead) L.pHead = L.pTail = p;
    else { p->pNext = L.pHead; L.pHead = p; }
}
void AddTail(LIST &L, NODE *p) {
    // PHẢI cập nhật pTail
    if (!L.pHead) L.pHead = L.pTail = p;
    else { L.pTail->pNext = p; L.pTail = p; }
}

DSLK — duyệt, tìm, thêm sau X

void Traverse(LIST L) {
    // template duyệt mọi bài xử lý DSLK
    for (NODE *p = L.pHead; p; p = p->pNext)
        cout << p->info << ' ';
}
NODE *searchNode(LIST L, int x) {
    for (NODE *p = L.pHead; p; p = p->pNext)
        if (p->info == x) return p;
    return NULL;
}
void add_Y_after_X(LIST &L, int Y, int X) {
    // Q là pTail thì pTail phải trỏ sang p
    NODE *Q = searchNode(L, X);
    if (Q == NULL) return;
    NODE *p = CreateNode(Y);
    p->pNext = Q->pNext;
    Q->pNext = p;
    if (L.pTail == Q) L.pTail = p;
}

DSLK — xóa đầu, xóa X

void RemoveHead(LIST &L) {
    // xóa hết → pTail = NULL, đừng quên
    if (!L.pHead) return;
    NODE *p = L.pHead; L.pHead = p->pNext; delete p;
    if (!L.pHead) L.pTail = NULL;
}
bool RemoveX(LIST &L, int x) {
    // giữ prev; x ở pTail → pTail = prev
    NODE *prev = NULL, *p = L.pHead;
    while (p && p->info != x) { prev = p; p = p->pNext; }
    if (!p) return false;
    if (!prev) L.pHead = p->pNext;
    else prev->pNext = p->pNext;
    if (p == L.pTail) L.pTail = prev;
    delete p; return true;
}

Đa thức — struct & Input

struct DONTHUC { float HeSo; int SoMu; };
struct PNODE { DONTHUC info; PNODE *pNext; };
struct DATHUC { PNODE *pHead, *pTail; };

PNODE *CreateNode(DONTHUC x) {
    PNODE *p = new PNODE;
    p->info = x; p->pNext = NULL; return p;
}
void AddTail(DATHUC &L, PNODE *p) {
    if (!L.pHead) L.pHead = L.pTail = p;
    else { L.pTail->pNext = p; L.pTail = p; }
}
void CreateEmptyDATHUC(DATHUC &poly) {
    poly.pHead = poly.pTail = NULL;
}
DONTHUC Input(float heso, int somu) {
    DONTHUC d; d.HeSo = heso; d.SoMu = somu;
    return d;
}
void Input(DATHUC &poly) {
    // đề cho: số mũ GIẢM dần, AddTail giữ thứ tự
    CreateEmptyDATHUC(poly);
    int n; cin >> n;
    for (int i = 0; i < n; i++) {
        float hs; int sm; cin >> hs >> sm;
        AddTail(poly, CreateNode(Input(hs, sm)));
    }
}

Đa thức — Sum (đề Câu 9)

DATHUC Sum(DATHUC poly1, DATHUC poly2) {
    // merge theo số mũ; hs == 0 thì BỎ đơn thức
    DATHUC kq; CreateEmptyDATHUC(kq);
    PNODE *p = poly1.pHead, *q = poly2.pHead;
    while (p && q) {
        if (p->info.SoMu > q->info.SoMu) {
            AddTail(kq, CreateNode(p->info)); p = p->pNext;
        } else if (p->info.SoMu < q->info.SoMu) {
            AddTail(kq, CreateNode(q->info)); q = q->pNext;
        } else {
            float hs = p->info.HeSo + q->info.HeSo;
            if (hs != 0)
                AddTail(kq, CreateNode(Input(hs, p->info.SoMu)));
            p = p->pNext; q = q->pNext;
        }
    }
    for (; p; p = p->pNext) AddTail(kq, CreateNode(p->info));
    for (; q; q = q->pNext) AddTail(kq, CreateNode(q->info));
    return kq;
}

BST — struct, chèn, tìm

struct TNODE { int key; TNODE *pLeft, *pRight; };
typedef TNODE *TREE;

void InsertNode(TREE &T, int x) {
    // x == T->key ⇒ KHÔNG chèn
    if (T == NULL) { T = new TNODE{x, NULL, NULL}; return; }
    if (x < T->key) InsertNode(T->pLeft, x);
    else if (x > T->key) InsertNode(T->pRight, x);
}
TNODE *SearchNode(TREE T, int x) {
    while (T && T->key != x)
        T = (x < T->key) ? T->pLeft : T->pRight;
    return T;
}

BST — 6 phép duyệt bằng một hàm

void Traverse(TREE T, const string &order) {
    // order = "NLR","LNR","LRN","NRL","RNL","RLN"
    if (!T) return;
    for (char c : order) {
        if (c == 'N') cout << T->key << ' ';
        else if (c == 'L') Traverse(T->pLeft, order);
        else Traverse(T->pRight, order);
    }
}

Nhớ: NLR tiền tự, LNR trung tự (tăng dần), LRN hậu tự; chữ R trước L = duyệt ngược (RNL cho khóa giảm dần).

BST — đếm, lá, tổng theo điều kiện

int CountNode(TREE T) {
    return T ? 1 + CountNode(T->pLeft)
               + CountNode(T->pRight) : 0;
}
int CountLeaf(TREE T) {
    if (!T) return 0;
    if (!T->pLeft && !T->pRight) return 1;
    return CountLeaf(T->pLeft) + CountLeaf(T->pRight);
}
int SumEven(TREE T) { // template: đổi điều kiện & phép cộng
    if (!T) return 0;
    return (T->key % 2 == 0 ? T->key : 0)
         + SumEven(T->pLeft) + SumEven(T->pRight);
}

BST — chiều cao, in mức k

int Height(TREE T) {
    // cây rỗng cao 0; node = 1 + max(2 con)
    if (!T) return 0;
    return 1 + max(Height(T->pLeft), Height(T->pRight));
}
void PrintLevel(TREE T, int k) {
    // mức gốc = 0; k == 0 thì in node hiện tại
    if (!T) return;
    if (k == 0) { cout << T->key << ' '; return; }
    PrintLevel(T->pLeft, k - 1);
    PrintLevel(T->pRight, k - 1);
}

BST — chẵn lớn nhất (đề Câu 10)

int EvenMax(TNODE *Root) {
    // R-N-L: chẵn ĐẦU TIÊN gặp là lớn nhất
    if (Root == NULL) return -1; // -1 lẻ: không nhầm với khóa chẵn
    int r = EvenMax(Root->pRight);
    if (r != -1) return r;
    if (Root->key % 2 == 0) return Root->key;
    return EvenMax(Root->pLeft);
}

BST xóa: đề chỉ yêu cầu vẽ tay — node 2 con thay bằng nhỏ nhất cây con phải (kể cả mirror: lớn nhất cây con trái). Xem B-Tree & Bảng băm cho phần vẽ tay B-Tree.
